   ---------------------------------------------------------------------
   ATENCIÓN: este script BORRA HotelDB si ya existe y la vuelve a crear.
 
   Convenciones
   - PK int IDENTITY (Auditoria usa bigint)
   - Dinero decimal(18,2), porcentajes decimal(5,2)
   - Fechas de hospedaje: date. Eventos: datetime2(0) en UTC (sufijo Utc)
   - Estados y tipos: tinyint con CHECK (ver catálogo de códigos abajo)
   - Borrado lógico con Activo en catálogos; FKs sin cascada
 
   Códigos de estados
   EstadoReservacion  1 PendientePago 2 Confirmada 3 EnCurso 4 Finalizada 5 Cancelada
   EstadoHabitacion   1 Disponible 2 Reservada 3 Ocupada 4 EnLimpieza 5 EnMantenimiento 6 Bloqueada
   EstadoEstadia      1 Activa 2 Finalizada
   EstadoPedido       1 Pendiente 2 Confirmado 3 EnPreparacion 4 Listo 5 EnCamino 6 Entregado 7 Cancelado
   ModalidadPedido    1 RoomService 2 Mesa 3 Piscina
   TipoCuenta         1 Hospedaje 2 Consumo
   EstadoCuenta       1 Abierta 2 Cerrada
   TipoCargo          1 Hospedaje 2 Pedido 3 LateCheckOut 4 Servicio 5 Ajuste 6 Penalizacion
   MetodoPago         1 Efectivo 2 Tarjeta 3 Transferencia
   EstadoPago         1 Aplicado 2 Revertido
   EstadoFactura      1 Emitida 2 Anulada
   TipoDocumento      1 Cedula 2 DNI 3 Pasaporte
   EstadoResMesa      1 Pendiente 2 Confirmada 3 Completada 4 Cancelada
   EstadoCorreo       1 Pendiente 2 Enviado 3 Fallido
   ===================================================================== */

CREATE TABLE dbo.Clientes (
    Id                   int IDENTITY(1,1) NOT NULL,

    FechaExpiracionUtc  datetime2(0)      NOT NULL,
    FechaUsoUtc         datetime2(0)      NULL,
    FechaRevocacionUtc  datetime2(0)      NULL,
    MotivoRevocacion    nvarchar(200)     NULL,
    ReemplazadoPorId    int               NULL,
    CONSTRAINT PK_RefreshTokens PRIMARY KEY (Id),

    CONSTRAINT FK_RefreshTokens_Usuarios FOREIGN KEY (UsuarioId) REFERENCES dbo.Usuarios (Id),
    CONSTRAINT FK_RefreshTokens_Reemplazo FOREIGN KEY (ReemplazadoPorId) REFERENCES dbo.RefreshTokens (Id),
    CONSTRAINT CK_RefreshTokens_Expiracion CHECK (FechaExpiracionUtc > FechaCreacionUtc)

GO
 
CREATE INDEX IX_RefreshTokens_Usuario ON dbo.RefreshTokens (UsuarioId);

CREATE TABLE dbo.TiposHabitacion (
    Id                      int IDENTITY(1,1) NOT NULL,

    Descripcion             nvarchar(500)  NOT NULL,
    CapacidadAdultos        tinyint        NOT NULL,
    CapacidadNinos          tinyint        NOT NULL,
    CapacidadMaxima         tinyint        NOT NULL,
    OcupacionBase           tinyint        NOT NULL,
    PrecioBasePorNoche      decimal(18,2)  NOT NULL,
    PrecioPersonaAdicional  decimal(18,2)  NOT NULL CONSTRAINT DF_TiposHabitacion_PrecioAdicional DEFAULT (0),
    Activo                  bit            NOT NULL CONSTRAINT DF_TiposHabitacion_Activo DEFAULT (1),
    CONSTRAINT PK_TiposHabitacion PRIMARY KEY (Id),
    CONSTRAINT UQ_TiposHabitacion_Nombre UNIQUE (Nombre),
    CONSTRAINT CK_TiposHabitacion_Adultos CHECK (CapacidadAdultos >= 1),
    CONSTRAINT CK_TiposHabitacion_Ocupacion CHECK (OcupacionBase >= 1 AND OcupacionBase <= CapacidadMaxima),
    CONSTRAINT CK_TiposHabitacion_Maxima CHECK (CapacidadMaxima <= CAST(CapacidadAdultos AS int) + CapacidadNinos),
    CONSTRAINT CK_TiposHabitacion_Precio CHECK (PrecioBasePorNoche > 0 AND PrecioPersonaAdicional >= 0)

-- Cabecera: fechas, cantidades y totales son el resumen de sus líneas
CREATE TABLE dbo.Reservaciones (
    Id                           int IDENTITY(1,1) NOT NULL,
    Codigo                       nvarchar(30)   NOT NULL,
    ClienteId                    int            NOT NULL,
    FechaInicio                  date           NOT NULL,
    FechaFin                     date           NOT NULL,
    Estado                       tinyint        NOT NULL CONSTRAINT DF_Reservaciones_Estado DEFAULT (1),
    CantidadAdultos              tinyint        NOT NULL,
    CantidadNinos                tinyint        NOT NULL CONSTRAINT DF_Reservaciones_Ninos DEFAULT (0),
    Subtotal                     decimal(18,2)  NOT NULL CONSTRAINT DF_Reservaciones_Subtotal DEFAULT (0),
    Descuento                    decimal(18,2)  NOT NULL CONSTRAINT DF_Reservaciones_Descuento DEFAULT (0),
    Impuestos                    decimal(18,2)  NOT NULL CONSTRAINT DF_Reservaciones_Impuestos DEFAULT (0),
    Total                        decimal(18,2)  NOT NULL CONSTRAINT DF_Reservaciones_Total DEFAULT (0),
    AnticipoRequerido            decimal(18,2)  NOT NULL CONSTRAINT DF_Reservaciones_Anticipo DEFAULT (0),
    FechaExpiracionPendienteUtc  datetime2(0)   NULL,
    Observaciones                nvarchar(1000) NULL,
    MotivoCancelacion            nvarchar(500)  NULL,
    CreadoPorUsuarioId           int            NOT NULL,
    FechaCreacionUtc             datetime2(0)   NOT NULL CONSTRAINT DF_Reservaciones_FechaCreacion DEFAULT (SYSUTCDATETIME()),
    RowVersion                   rowversion     NOT NULL,
    CONSTRAINT PK_Reservaciones PRIMARY KEY (Id),
    CONSTRAINT UQ_Reservaciones_Codigo UNIQUE (Codigo),
    CONSTRAINT FK_Reservaciones_Clientes FOREIGN KEY (ClienteId) REFERENCES dbo.Clientes (Id),
    CONSTRAINT FK_Reservaciones_Usuarios FOREIGN KEY (CreadoPorUsuarioId) REFERENCES dbo.Usuarios (Id),
    CONSTRAINT CK_Reservaciones_Fechas CHECK (FechaInicio < FechaFin),
    CONSTRAINT CK_Reservaciones_Estado CHECK (Estado BETWEEN 1 AND 5),
    CONSTRAINT CK_Reservaciones_Adultos CHECK (CantidadAdultos >= 1),
    CONSTRAINT CK_Reservaciones_Montos CHECK (Subtotal >= 0 AND Descuento >= 0 AND Impuestos >= 0
                                              AND Total >= 0 AND AnticipoRequerido >= 0 AND AnticipoRequerido <= Total),
    CONSTRAINT CK_Reservaciones_Expiracion CHECK (Estado <> 1 OR FechaExpiracionPendienteUtc IS NOT NULL),
    CONSTRAINT CK_Reservaciones_Cancelacion CHECK (Estado <> 5 OR MotivoCancelacion IS NOT NULL)
);
GO
 
CREATE INDEX IX_Reservaciones_Cliente_Estado ON dbo.Reservaciones (ClienteId, Estado) INCLUDE (FechaInicio, FechaFin, Codigo);
CREATE INDEX IX_Reservaciones_Estado_Expiracion ON dbo.Reservaciones (Estado, FechaExpiracionPendienteUtc);
CREATE INDEX IX_Reservaciones_Fechas ON dbo.Reservaciones (FechaInicio, FechaFin) INCLUDE (Estado);
GO

/* Una fila por noche ocupada. El índice único filtrado (HabitacionId, Fecha)
   impide que dos reservas activas tomen la misma noche: la API recibe el error
   2601/2627 y responde 409. Intervalo [entrada, salida): del 10 al 13 = noches 10, 11, 12.
   Al cancelar o vencer la reserva, sus noches pasan a Activa = 0. */
CREATE TABLE dbo.ReservacionNoches (
    Id                        int IDENTITY(1,1) NOT NULL,
    ReservacionHabitacionId   int            NOT NULL,
    HabitacionId              int            NOT NULL,
    Fecha                     date           NOT NULL,
    TarifaBase                decimal(18,2)  NOT NULL,
    AjusteTarifa              decimal(18,2)  NOT NULL CONSTRAINT DF_ResNoches_Ajuste DEFAULT (0),
    TarifaFinal               decimal(18,2)  NOT NULL,
    Activa                    bit            NOT NULL CONSTRAINT DF_ResNoches_Activa DEFAULT (1),
    CONSTRAINT PK_ReservacionNoches PRIMARY KEY (Id),
    CONSTRAINT FK_ResNoches_ResHab FOREIGN KEY (ReservacionHabitacionId) REFERENCES dbo.ReservacionHabitaciones (Id),
    CONSTRAINT FK_ResNoches_Habitaciones FOREIGN KEY (HabitacionId) REFERENCES dbo.Habitaciones (Id),
    CONSTRAINT UQ_ResNoches_Linea_Fecha UNIQUE (ReservacionHabitacionId, Fecha),
    CONSTRAINT CK_ResNoches_Tarifas CHECK (TarifaBase >= 0 AND TarifaFinal >= 0)
);
GO
 
CREATE UNIQUE INDEX UX_ResNoches_Habitacion_Fecha_Activa ON dbo.ReservacionNoches (HabitacionId, Fecha) WHERE Activa = 1;
GO

-- Una estadía por habitación reservada

    CheckInRealUtc            datetime2(0)  NOT NULL,
    CheckOutRealUtc           datetime2(0)  NULL,
    UsuarioCheckInId          int           NOT NULL,
    UsuarioCheckOutId         int           NULL,
    Estado                    tinyint       NOT NULL CONSTRAINT DF_Estadias_Estado DEFAULT (1),
    RowVersion                rowversion    NOT NULL,

    CONSTRAINT UQ_Estadias_ResHab UNIQUE (ReservacionHabitacionId),

    CONSTRAINT FK_Estadias_UsuarioCheckOut FOREIGN KEY (UsuarioCheckOutId) REFERENCES dbo.Usuarios (Id),
    CONSTRAINT CK_Estadias_Estado CHECK (
        (Estado = 1 AND CheckOutRealUtc IS NULL AND UsuarioCheckOutId IS NULL)

    CONSTRAINT CK_Estadias_Fechas CHECK (CheckOutRealUtc IS NULL OR CheckOutRealUtc > CheckInRealUtc)

CREATE INDEX IX_Estadias_Estado ON dbo.Estadias (Estado) INCLUDE (ReservacionHabitacionId, CheckInRealUtc);

/* =====================================================================
   CARGOS, PAGOS Y FACTURACIÓN
   Saldo de una cuenta = SUM(CargosCuenta.Total) - (SUM(Pagos aplicados) - SUM(ReversosPago))
   ===================================================================== */
 
/* Los cargos no se editan ni se borran. Una corrección es otro cargo con
   PrecioUnitario negativo que apunta al original en CargoRevertidoId. */
CREATE TABLE dbo.CargosCuenta (
    Id                int IDENTITY(1,1) NOT NULL,
    CuentaId          int            NOT NULL,
    Tipo              tinyint        NOT NULL,
    EstadiaId         int            NULL,
    PedidoId          int            NULL,
    Concepto          nvarchar(200)  NOT NULL,
    Cantidad          decimal(10,2)  NOT NULL,
    PrecioUnitario    decimal(18,2)  NOT NULL,
    Descuento         decimal(18,2)  NOT NULL CONSTRAINT DF_Cargos_Descuento DEFAULT (0),
    ImpuestoId        int            NULL,
    Impuestos         decimal(18,2)  NOT NULL CONSTRAINT DF_Cargos_Impuestos DEFAULT (0),
    Total             decimal(18,2)  NOT NULL,
    FechaUtc          datetime2(0)   NOT NULL CONSTRAINT DF_Cargos_Fecha DEFAULT (SYSUTCDATETIME()),
    UsuarioId         int            NOT NULL,
    CargoRevertidoId  int            NULL,
    MotivoAjuste      nvarchar(300)  NULL,
    AutorizadoPorId   int            NULL,
    CONSTRAINT PK_CargosCuenta PRIMARY KEY (Id),
    CONSTRAINT FK_Cargos_Cuentas FOREIGN KEY (CuentaId) REFERENCES dbo.Cuentas (Id),
    CONSTRAINT FK_Cargos_Estadias FOREIGN KEY (EstadiaId) REFERENCES dbo.Estadias (Id),
    CONSTRAINT FK_Cargos_Pedidos FOREIGN KEY (PedidoId) REFERENCES dbo.PedidosRestaurante (Id),
    CONSTRAINT FK_Cargos_Impuestos FOREIGN KEY (ImpuestoId) REFERENCES dbo.Impuestos (Id),
    CONSTRAINT FK_Cargos_Usuarios FOREIGN KEY (UsuarioId) REFERENCES dbo.Usuarios (Id),
    CONSTRAINT FK_Cargos_Autorizador FOREIGN KEY (AutorizadoPorId) REFERENCES dbo.Usuarios (Id),
    CONSTRAINT FK_Cargos_Revertido FOREIGN KEY (CargoRevertidoId) REFERENCES dbo.CargosCuenta (Id),
    CONSTRAINT CK_Cargos_Tipo CHECK (Tipo BETWEEN 1 AND 6),
    CONSTRAINT CK_Cargos_Cantidad CHECK (Cantidad > 0),
    CONSTRAINT CK_Cargos_Descuento CHECK (Descuento >= 0),
    CONSTRAINT CK_Cargos_Pedido CHECK (Tipo <> 2 OR PedidoId IS NOT NULL),
    CONSTRAINT CK_Cargos_Ajuste CHECK (Tipo <> 5 OR (MotivoAjuste IS NOT NULL AND AutorizadoPorId IS NOT NULL)),
    CONSTRAINT CK_Cargos_Reverso CHECK (CargoRevertidoId IS NULL OR MotivoAjuste IS NOT NULL)
);
GO
 
-- Un cargo solo se puede revertir una vez
CREATE UNIQUE INDEX UX_Cargos_Revertido ON dbo.CargosCuenta (CargoRevertidoId) WHERE CargoRevertidoId IS NOT NULL;
CREATE INDEX IX_Cargos_Cuenta ON dbo.CargosCuenta (CuentaId) INCLUDE (Total, Tipo, FechaUtc);
CREATE INDEX IX_Cargos_Pedido ON dbo.CargosCuenta (PedidoId) WHERE PedidoId IS NOT NULL;
GO

INSERT INTO dbo.HabitacionAmenidades (HabitacionId, AmenidadId)
SELECT h.Id, a.Id
FROM dbo.Habitaciones h
JOIN dbo.TiposHabitacion t ON t.Id = h.TipoHabitacionId
JOIN dbo.Amenidades a ON
     a.Nombre IN (N'Aire acondicionado', N'Wi-Fi', N'Caja fuerte')
  OR (t.Nombre IN (N'Suite Ejecutiva', N'Suite Presidencial') AND a.Nombre IN (N'Cama king', N'Minibar', N'Balcón'))
  OR (t.Nombre = N'Suite Presidencial' AND a.Nombre IN (N'Jacuzzi', N'Vista al mar'))
  OR (t.Nombre = N'Familiar' AND a.Nombre IN (N'Cama king', N'Balcón'));
GO
